cmake_minimum_required(VERSION 3.10)
project(migration_matrix)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Add include directory
include_directories(${PROJECT_SOURCE_DIR}/include)

# Collect source files (excluding main function)
set(SOURCES
    src/deathages.cpp
    src/DataLoader.cpp
    src/fertility.cpp
    src/migration.cpp
    src/utils.cpp
    src/validate.cpp
)

# Create main executable
add_executable(get_migration_matrix 
    ${SOURCES}
    src/Get_migration_matrix.cpp
)

# Create test executable (without main function)
add_executable(test_migration 
    ${SOURCES}
    test/Test_migration.cpp
)

# Set output directory
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/build)
